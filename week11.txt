#dnorm
x<-seq(20,80,by=1)
y<-dnorm(x,mean = 50,sd=10)
plot(x,y,type = "p",pch=19,col="blue",main = "dnorm",xlab = "marks",ylab = "density")
model<-lm(y~x)
abline(model,col="red",lwd=1)
summary(model)
#pnorm
x<-seq(20,80,by=1)
y<-pnorm(x,mean = 50,sd=10)
plot(x,y,type = "p",pch=19,col="blue",main = "commulative probability(pnorm)",xlab = "marks",ylab = "probability")
model<-lm(y~x)
abline(model,col="red",lwd=2)
summary(model)
#qnorm
p<-seq(0.01,0.99,0.05)

#rnorm
set.seed(100)
y<-rnorm(50,mean = 50,sd=10)
x<-1:50
plot(x,y,pch=19,col="blue",main = "random normal data(rnorm)",xlab = "observations",ylab = "marks")
model<-lm(y~x)
abline(model,col="red",lwd=2)
summary(model)
